function buildTasks(key){
 const w=D.workouts[key],wNum=Number(state.week)||1,reduce=wNum<=2||wNum===7,t=[];
 w.blocks.forEach((b,bi)=>{
  if(b.type==='warmup')t.push({kind:'warmup',name:b.name,duration:b.seconds,cue:b.cue,next:blockNext(w.blocks,bi)});
  else if(b.type==='exercise'){
   const sets=reduce?Math.min(2,b.sets):b.sets;
   for(let s=1;s<=sets;s++){
    t.push({kind:'work',name:b.name,duration:45,cue:b.cue,reps:b.reps,set:s,sets,exercise:b.name});
    if(s<sets)t.push({kind:'rest',name:`Rest after ${b.name}`,duration:b.rest,cue:`Breathe, record the set, and prepare for set ${s+1}.`,next:`${b.name} set ${s+1}`});
   }
   if(bi<w.blocks.length-1)t.push({kind:'transition',name:'Change station',duration:20,cue:`Move efficiently to ${blockName(w.blocks[bi+1])}.`,next:blockName(w.blocks[bi+1])});
  }else{
   const rounds=reduce?Math.min(2,b.rounds):b.rounds;
   for(let r=1;r<=rounds;r++){
    b.items.forEach((it,ii)=>{
     t.push({kind:'work',name:it.name,duration:35,cue:it.cue,reps:it.reps,set:r,sets:rounds,exercise:it.name,circuit:b.name});
     if(ii<b.items.length-1)t.push({kind:'transition',name:'Circuit transition',duration:10,cue:`Move directly to ${b.items[ii+1].name}.`,next:b.items[ii+1].name});
    });
    if(r<rounds)t.push({kind:'rest',name:`Rest after circuit round ${r}`,duration:b.rest,cue:`Recover, then repeat ${b.name}.`,next:`${b.name} round ${r+1}`});
   }
  }
 });
 return t;
}
function blockName(b){return b?.name||'the next block'}
function blockNext(arr,i){return blockName(arr[i+1])}
async function startCoach(key){
 if(!D.workouts[key])return;
 const tasks=buildTasks(key),now=new Date(),realStop=new Date(now);realStop.setHours(5,18,0,0);const morning=now.getHours()>=3&&now<realStop;
 coach={key,tasks,index:0,remaining:tasks[0].duration,elapsed:0,paused:false,voice:state.settings.voice!==false,lastTick:Date.now(),deadline:morning?realStop.getTime():Date.now()+48*60*1000,announced:new Set(),logged:false};
 $('#coach').classList.remove('hidden');document.body.style.overflow='hidden';
 $('#coach-day').textContent=D.workouts[key].title;$('#voice-toggle').textContent=coach.voice?'Voice On':'Voice Off';
 try{if('wakeLock'in navigator)wakeLock=await navigator.wakeLock.request('screen')}catch{}
 showCoachTask(true);coach.timer=setInterval(coachTick,250);speak(`Hercules 5 30. ${D.workouts[key].title}. You have a hard stop at 5 18. Begin ${tasks[0].name}. ${tasks[0].cue}`);
}
function coachTick(){if(!coach||coach.paused)return;const now=Date.now(),delta=(now-coach.lastTick)/1000;coach.lastTick=now;coach.remaining-=delta;coach.elapsed+=delta;if(now>=coach.deadline){finishCoach(true);return}announceThresholds();if(coach.remaining<=0)advanceCoach(1);updateCoachUI()}
function announceThresholds(){const t=coach.tasks[coach.index],r=Math.ceil(coach.remaining);if(t.kind==='rest'){[30,10,3].forEach(n=>{const k=`${coach.index}-${n}`;if(r<=n&&!coach.announced.has(k)){coach.announced.add(k);speak(n===3?'Three, two, one. Prepare.':`${n} seconds remaining.`)}})}else if(t.kind==='work'){[20,5].forEach(n=>{const k=`${coach.index}-${n}`;if(r<=n&&!coach.announced.has(k)){coach.announced.add(k);speak(n===5?'Five seconds. Finish with clean form.':t.cue)}})}}
function advanceCoach(dir){if(!coach)return;coach.index=Math.max(0,Math.min(coach.tasks.length,coach.index+dir));if(coach.index>=coach.tasks.length){finishCoach(false);return}coach.remaining=coach.tasks[coach.index].duration;coach.lastTick=Date.now();coach.logged=false;showCoachTask(true)}
function showCoachTask(announce=false){const t=coach.tasks[coach.index];if(!t)return;$('#coach-phase').textContent=t.kind==='work'?`SET ${t.set} OF ${t.sets} · ${t.reps} REPS`:t.kind.toUpperCase();$('#coach-task').textContent=t.name;$('#coach-cue').textContent=t.cue;$('#coach-next').textContent=`Next: ${coach.tasks[coach.index+1]?.name||'Workout complete'}`;$('#set-log').classList.toggle('hidden',t.kind!=='work');if(t.kind==='work'){const last=[...state.sets].reverse().find(x=>x.exercise===t.exercise);$('#log-weight').value=last?.weight||'';$('#log-reps').value='';$('#log-rir').value=Number(state.week)<=2||Number(state.week)===7?'3':'2'}if(announce)speak(taskSpeech(t));updateCoachUI()}
function taskSpeech(t){if(t.kind==='work')return `${t.name}. Set ${t.set} of ${t.sets}. Target ${t.reps} reps. ${t.cue}`;if(t.kind==='rest')return `Rest ${t.duration} seconds. ${t.cue}`;return `${t.name}. ${t.cue}`}
function updateCoachUI(){if(!coach)return;const t=coach.tasks[coach.index];$('#coach-clock').textContent=clock(Math.ceil(coach.remaining));$('#coach-elapsed').textContent=`${clock(coach.elapsed)} / 48:00`;$('#coach-bar').style.width=`${Math.min(100,coach.elapsed/(48*60)*100)}%`;$('#coach-pause').textContent=coach.paused?'Resume':'Pause';$('#voice-toggle').textContent=coach.voice?'Voice On':'Voice Off'}

function releaseCoach(){clearInterval(coach?.timer);coach=null;$('#coach').classList.add('hidden');document.body.style.overflow='';if(wakeLock){wakeLock.release().catch(()=>{});wakeLock=null}render()}
