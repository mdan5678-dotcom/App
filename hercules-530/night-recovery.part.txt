function isNightRecoveryItem(x){
 const name=String(x?.name||'').toLowerCase();
 return name.includes('magnesium bisglycinate')||name.includes('collagen peptides');
}
function nightRecoveryItems(){
 const list=(D.supplements?.training||[]).filter(isNightRecoveryItem);
 const magnesium=list.find(x=>String(x.name).toLowerCase().includes('magnesium'));
 const collagen=list.find(x=>String(x.name).toLowerCase().includes('collagen'));
 return [
  magnesium||{time:'30-60 min before bed',name:'Magnesium bisglycinate',dose:'1 scoop / about 200 mg elemental',note:'Take nightly with water. Verify the elemental-magnesium amount on the bottle label and reduce or stop for diarrhea.'},
  collagen||{time:'30-60 min before bed',name:'Collagen peptides',dose:'1 scoop',note:'Take nightly with water for connective-tissue support. Do not count collagen as complete muscle-building protein.'}
 ];
}
['training','rest'].forEach(mode=>{
 const items=D.meals?.[mode]?.items;
 if(Array.isArray(items)&&!items.some(x=>String(x.name||'').toLowerCase().includes('night recovery'))){
  items.push({time:'30-60 min before bed',name:'Night recovery',food:'Thorne Magnesium Bisglycinate: 1 scoop (about 200 mg elemental; verify the bottle label). Momentous Collagen Peptides: 1 scoop. Take both every night with water.'});
 }
});

 const otherSupplements=allSupplements.filter(x=>!isNightRecoveryItem(x));
 const nightItems=nightRecoveryItems();
 APP.innerHTML=`<section class="hero"><p class="eyebrow">${train?'TODAY\'S MISSION':'RECOVERY BUILDS MUSCLE'}</p><h1>${esc(info.title)}</h1><p>${train?esc(w.focus)+'.
